TEST Event (command line):

$ts = [int](Get-Date -UFormat %s)

$body = @{
    device_id = "esp32_001"
    event_type = "door_open"
    count = 1
    timestamp = $ts
} | ConvertTo-Json

Invoke-RestMethod `
  -Uri http://localhost:8000/events `
  -Method POST `
  -ContentType "application/json" `
  -Body $body
